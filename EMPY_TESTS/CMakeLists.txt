file(GLOB sources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(EmpyTests ${sources})
configure_project_target(EmpyTests)
target_link_libraries(EmpyTests PRIVATE ${EMPY_LIB_CLIENT})
empy_stage_runtime(EmpyTests)
target_compile_definitions(EmpyTests PRIVATE EMPY_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

foreach(suite protocol runtime security integration cluster adversarial correlation monitoring)
    add_test(NAME ${suite} COMMAND EmpyTests ${suite})
    set_tests_properties(${suite} PROPERTIES TIMEOUT 30)
endforeach()
add_test(NAME recovery COMMAND EmpyTests recovery)
set_tests_properties(recovery PROPERTIES TIMEOUT 120)
